```python
"""
SPICE Netlist Exporter for circuit-synth

Converts circuit-synth Circuit objects to SPICE netlist format.
Maps KiCad symbols to SPICE components.
"""

class SpiceExporter:
    # Component mapping from KiCad symbols to SPICE syntax
    SPICE_COMPONENT_MAP = {
        "Device:R": "R{ref} {pin1} {pin2} {value}",
        "Device:C": "C{ref} {pin1} {pin2} {value}",
        "Device:L": "L{ref} {pin1} {pin2} {value}",
        # Add more as needed
    }
    
    def __init__(self, circuit):
        self.circuit = circuit
        
    def generate_netlist(self, include_analysis=True) -> str:
        """Generate complete SPICE netlist"""
        lines = [
            f"* {self.circuit.name}",
            f"* Generated by circuit-synth",
            "",
            f".title {self.circuit.name}",
            ""
        ]
        
        # Add components
        lines.append("* Components")
        for comp in self.circuit.get_components():
            spice_line = self._component_to_spice(comp)
            if spice_line:
                lines.append(spice_line)
        
        # Add voltage sources (for analysis)
        lines.append("")
        lines.append("* Voltage sources")
        # TODO: Auto-detect input nodes and add voltage sources
        
        if include_analysis:
            lines.extend([
                "",
                "* Analysis commands",
                ".DC VIN 0 5 0.1",
                ".AC DEC 10 0.1 1MEG",
                "",
                "* Output requests", 
                ".PRINT DC V(VOUT)",
                ".PRINT AC VDB(VOUT) VP(VOUT)"
            ])
        
        lines.append("")
        lines.append(".END")
        
        return "\n".join(lines)
    
    def _component_to_spice(self, component) -> str:
        """Convert single component to SPICE syntax"""
        symbol = component.symbol
        if symbol not in self.SPICE_COMPONENT_MAP:
            # Skip unsupported components with warning
            return f"* WARNING: Unsupported component {symbol}"
        
        template = self.SPICE_COMPONENT_MAP[symbol]
        
        # Get pin connections
        pins = self._get_component_pins(component)
        
        return template.format(
            ref=component.ref,
            pin1=pins[0],
            pin2=pins[1] if len(pins) > 1 else "0",
            value=component.value or "1"
        )
```

```python
@classmethod
def from_spice(cls, netlist: str) -> 'CircuitSimulator':
    """
    Create simulator from SPICE netlist.
    
    Args:
        netlist: Complete SPICE netlist string
        
    Returns:
        CircuitSimulator: Ready for analysis
    """
    # Write netlist to temporary file
    import tempfile
    with tempfile.NamedTemporaryFile(mode='w', suffix='.spice', delete=False) as f:
        f.write(netlist)
        netlist_path = f.name
    
    # Use existing PySpice integration to load netlist
    from PySpice.Spice.Netlist import Circuit as SpiceCircuit
    spice_circuit = SpiceCircuit.from_netlist_file(netlist_path)
    
    # Create simulator instance
    simulator = cls()
    simulator._spice_circuit = spice_circuit
    simulator._netlist_path = netlist_path
    
    return simulator
```

```python
def test_voltage_divider_integration():
    """Test complete workflow with real circuit-synth and circuit-simulation"""
    
    # Create circuit in circuit-synth
    from circuit_synth import circuit, Component, Net
    
    @circuit(name="test_voltage_divider")
    def voltage_divider():
        r1 = Component("Device:R", ref="R", value="10k")
        r2 = Component("Device:R", ref="R", value="10k")
        
        vin = Net('VIN')
        vout = Net('VOUT')
        gnd = Net('GND')
        
        r1[1] += vin
        r1[2] += vout
        r2[1] += vout
        r2[2] += gnd
        
        return circuit
    
    # Export to SPICE
    circuit = voltage_divider()
    netlist = circuit.to_spice()
    
    assert "R1" in netlist
    assert "R2" in netlist
    assert "10k" in netlist
    
    # Import to circuit-simulation
    from circuit_simulation import Simulator
    sim = Simulator.from_spice(netlist)
    
    # Run analysis
    dc_result = sim.dc_analysis()
    
    # Check results (voltage divider should give VIN/2)
    vout_at_4v = dc_result.get_voltage("VOUT", vin=4.0)
    assert abs(vout_at_4v - 2.0) < 0.1  # Within 10% tolerance
```
